// =========================================================
// BULLBORN — règles de sécurité Firestore
// -----------------------------------------------------------
// À coller dans : Firebase Console -> Firestore Database ->
// onglet "Règles" -> remplace tout le contenu -> "Publier".
//
// Ce que ça autorise :
//  - N'importe qui peut CRÉER une commande (le checkout du site)
//  - N'importe qui peut LIRE une commande s'il connaît déjà son
//    CODE exact (pour suivi.html) — mais personne ne peut lister
//    toutes les commandes sans être connecté
//  - Seul le staff connecté (compte créé dans Authentication)
//    peut voir la LISTE de toutes les commandes et changer leur
//    statut (livreur.html)
// =========================================================
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /orders/{orderId} {
      allow get:    if true;
      allow list:   if request.auth != null;
      allow create: if true;
      allow update: if request.auth != null;
      allow delete: if false;
    }
  }
}
